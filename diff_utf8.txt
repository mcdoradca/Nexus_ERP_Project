+async function generateClaidLifestyle(imageBase64, sourceImageUrl, ean, imageIndex = 0) {
+    const claidKey = process.env.CLAID_API_KEY;
+    if (!claidKey || claidKey === "TBD") {
+        throw new Error("Brak prawid┼éowego klucza CLAID_API_KEY w .env. Skontaktuj si─Ö z administratorem.");
+    }
+
+    let inputForAiEdit = sourceImageUrl || imageBase64;
+
+    // 1. Je┼Ťli z frontendu przychodzi Base64 (nawet jako sourceImageUrl), uploadujemy do Claid na tymczasowy kube┼éek
+    if (inputForAiEdit && inputForAiEdit.startsWith('data:image')) {
+        console.log("[AiService] Otrzymano Base64. Upload do Claid Upload API...");
+        const base64Data = inputForAiEdit.replace(/^data:image\/\w+;base64,/, "");
+        const buffer = Buffer.from(base64Data, 'base64');
+        
+        const FormData = require('form-data');
+        const form = new FormData();
+        form.append('file', buffer, { filename: 'upload.jpg', contentType: 'image/jpeg' });
+        form.append('data', JSON.stringify({})); 
+
+        try {
+            const uploadRes = await axios.post('https://api.claid.ai/v1/image/edit/upload', form, {
+                headers: {
+                    'Authorization': `Bearer ${claidKey}`,
+                    ...form.getHeaders()
+                }
+            });
+
+            if (uploadRes.data?.data?.output?.tmp_url) {
+                inputForAiEdit = uploadRes.data.data.output.tmp_url;
+                console.log("[AiService] Sukces uploadu, tmp_url:", inputForAiEdit);
+            } else {
+                throw new Error("Nieprawid┼éowa odpowied┼║ z endpointu uploadu Claid.");
+            }
+        } catch (err) {
+            console.error("[AiService] B┼é─ůd uploadu Base64 do Claid:", err.response?.data || err.message);
+            throw new Error("B┼é─ůd podczas wgrywania pliku bazowego do serwer├│w Claid.");
+        }
+    }
+
+    if (!inputForAiEdit) {
+        throw new Error("Brak wej┼Ťciowego obrazu (imageBase64 ani sourceImageUrl).");
+    }
+
+    // 2. Generowanie inteligentnego promptu w Gemini
+    // Nazw─Ö produktu wyci─ůgamy w idealnym scenariuszu z bazy na podstawie EAN, ale dla uproszczenia rzutujemy na generyka
+    const prompt = await generateClaidLifestylePrompt(`Product EAN: ${ean}`, imageIndex);
+    console.log(`[AiService] Claid AI Edit Prompt [${imageIndex}]:`, prompt);
+
+    // 3. Generowanie Sceny (AI Photoshoot) z zachowaniem produktu
+    const aiScenePayload = {
+        object: {
+            image_url: inputForAiEdit,
+            placement_type: "original"
+        },
+        scene: {
+            model: "v2-beta",
+            prompt: prompt,
+            preference: "best"
+        },
+        output: {
+            number_of_images: 1,
+            format: "jpeg"
+        }
+    };
+
+    console.log("[AiService] Wys┼éanie zlecenia Scene Create (AI Photoshoot)...");
+    let sceneTmpUrl;
+    try {
+        const aiSceneRes = await axios.post('https://api.claid.ai/v1/scene/create', aiScenePayload, {
+            headers: {
+                'Authorization': `Bearer ${claidKey}`,
+                'Content-Type': 'application/json'
+            }
+        });
+        sceneTmpUrl = aiSceneRes.data?.data?.output?.[0]?.tmp_url;
+    } catch (err) {
+        console.error("[AiService] B┼é─ůd inicjalizacji zlecenia Claid Scene Create:", err.response?.data || err.message);
+        const errData = err.response?.data;
+        if (errData?.error_type === 'billing') {
+            throw new Error("Brak ┼Ťrodk├│w na koncie Claid API (Not enough API credits). Do┼éaduj konto Claid.");
+        }
+        throw new Error("Nie uda┼éo si─Ö utworzy─ç zlecenia Claid Scene Create. Pow├│d: " + (errData?.error_message || err.message));
+    }
+
+    if (!sceneTmpUrl) throw new Error("Brak URL wygenerowanej sceny od Claid API.");
+
+    // 4. Skalowanie i polerowanie (Prime / Smart Resize) - odblokowuje 4K i ultra-realistyczn─ů sk├│r─Ö
+    console.log("[AiService] Wys┼éanie zlecenia Image Edit (Upscale & Polish)...");
+    let finalTmpUrl = sceneTmpUrl;
+    try {
+        const aiEditPayload = {
+            input: sceneTmpUrl,
+            operations: {
+                restorations: {
+                    polish: true,
+                    upscale: "smart_resize"
+                },
+                resizing: {
+                    width: "200%",
+                    height: "200%",
+                    fit: "bounds"
+                }
+            },
+            output: {
+                format: "jpeg"
+            }
+        };
+
+        const aiEditRes = await axios.post('https://api.claid.ai/v1/image/edit', aiEditPayload, {
+            headers: {
+                'Authorization': `Bearer ${claidKey}`,
+                'Content-Type': 'application/json'
+            }
+        });
+        
+        const polishedUrl = aiEditRes.data?.data?.output?.tmp_url;
+        if (polishedUrl) {

+        console.error("[AiService] B┼é─ůd upscalingu (zapasowo u┼╝yto obrazu ze Scene Create):", err.response?.data || err.message);

+    }
